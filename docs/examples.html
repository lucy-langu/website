22 Sqlite

22_sqlite.lucy

import fs
import sqlite

let db = sqlite.open "lucy-example.sqlite"

db.execute(
    "CREATE TABLE IF NOT EXISTS users (id INTEGER PRIMARY KEY, name TEXT, age INTEGER)"
)

db.execute(
    "INSERT INTO users (name, age) VALUES (?, ?)",
    ["Lucy", 1]
)

let rows = db.query(
    "SELECT id, name, age FROM users WHERE age >= ?",
    [1]
)

print rows
print db.last_insert_id()
print db.changes()

db.begin()
db.execute("INSERT INTO users (name, age) VALUES (?, ?)", ["Nima", 20])
db.rollback()

let statement = db.prepare("SELECT name FROM users WHERE age >= ?")
statement.bind([1])
print statement.query()
statement.close()
db.close()

fs.remove "lucy-example.sqlite"

25 Builtin Type Extensions

25_builtin_type_extensions.lucy

# Built-in types can be extended directly; no import is required.
class String {
    func shout() {
        return self.upper() + "!"
    }
}

class Int {
    func doubled() {
        return self * 2
    }
}

println "Lucy".shout()
println 21.doubled()
